create table if not exists public.hl_properties (
  id text primary key,
  data jsonb not null,
  updated_at timestamptz default now()
);

create table if not exists public.hl_staff_accounts (
  id text primary key,
  data jsonb not null,
  updated_at timestamptz default now()
);

create table if not exists public.hl_customers (
  id text primary key,
  data jsonb not null,
  updated_at timestamptz default now()
);

create table if not exists public.hl_app_settings (
  id text primary key,
  data jsonb not null,
  updated_at timestamptz default now()
);

alter table public.hl_properties enable row level security;
alter table public.hl_staff_accounts enable row level security;
alter table public.hl_customers enable row level security;
alter table public.hl_app_settings enable row level security;

drop policy if exists "hl_properties_all" on public.hl_properties;
create policy "hl_properties_all" on public.hl_properties for all using (true) with check (true);

drop policy if exists "hl_staff_all" on public.hl_staff_accounts;
create policy "hl_staff_all" on public.hl_staff_accounts for all using (true) with check (true);

drop policy if exists "hl_customers_all" on public.hl_customers;
create policy "hl_customers_all" on public.hl_customers for all using (true) with check (true);

drop policy if exists "hl_app_settings_all" on public.hl_app_settings;
create policy "hl_app_settings_all" on public.hl_app_settings for all using (true) with check (true);

alter publication supabase_realtime add table public.hl_properties;
